Check health:
  kubectl -n {{ .Release.Namespace }} get pods -l app.kubernetes.io/instance={{ .Release.Name }}
  kubectl -n {{ .Release.Namespace }} exec -it {{ include "redis-sentinel.fullname" . }}-0 -c sentinel -- \
    redis-cli -p {{ .Values.sentinel.port }} sentinel master {{ .Values.redis.masterGroupName }}
  kubectl -n {{ .Release.Namespace }} exec -it {{ include "redis-sentinel.fullname" . }}-0 -c sentinel -- \
    redis-cli -p {{ .Values.sentinel.port }} sentinel get-master-addr-by-name {{ .Values.redis.masterGroupName }}

  kubectl -n {{ .Release.Namespace }} delete pod <current-master-pod>
