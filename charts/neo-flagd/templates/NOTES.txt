Checks worth running once after the first sync:

  kubectl -n {{ .Release.Namespace }} logs deploy/{{ include "neo-flagd.fullname" . }} | grep -i flyway

  # Readiness includes the snapshot indicator — this is what dequeues a replica whose
  # refresh loop has stalled. It must list flagSnapshots.
  kubectl -n {{ .Release.Namespace }} exec deploy/{{ include "neo-flagd.fullname" . }} -- \
    wget -qO- localhost:8080/actuator/health/readiness
